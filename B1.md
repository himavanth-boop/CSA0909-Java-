B1.1 – Implement File Storage Operations (3 Marks)
Java Code – BusPassFileStore.java
import java.io.*;
import java.util.*;

class BusPass {
    String passId, registerNumber, name;
    int routeNumber;
    String boardingStop, validityDate;
    double amountPaid;

    BusPass(String passId, String registerNumber, String name,
            int routeNumber, String boardingStop,
            String validityDate, double amountPaid) {
        this.passId = passId;
        this.registerNumber = registerNumber;
        this.name = name;
        this.routeNumber = routeNumber;
        this.boardingStop = boardingStop;
        this.validityDate = validityDate;
        this.amountPaid = amountPaid;
    }

    public String toString() {
        return passId + "," + registerNumber + "," + name + ","
                + routeNumber + "," + boardingStop + ","
                + validityDate + "," + amountPaid;
    }
}

public class BusPassFileStore {

    static final String FILE = "passes.txt";

    static void addPass(BusPass pass) throws IOException {
        try (BufferedWriter writer =
                     new BufferedWriter(new FileWriter(FILE, true))) {
            writer.write(pass.toString());
            writer.newLine();
        }
    }

    static BusPass findPass(String id) throws IOException {
        try (BufferedReader reader =
                     new BufferedReader(new FileReader(FILE))) {

            String line;

            while ((line = reader.readLine()) != null) {
                String[] data = line.split(",", -1);

                if (data.length != 7)
                    continue;

                if (data[0].equals(id)) {
                    return new BusPass(
                            data[0], data[1], data[2],
                            Integer.parseInt(data[3]),
                            data[4], data[5],
                            Double.parseDouble(data[6]));
                }
            }
        }
        return null;
    }

    static void listByRoute(int route) throws IOException {
        try (BufferedReader reader =
                     new BufferedReader(new FileReader(FILE))) {

            String line;

            while ((line = reader.readLine()) != null) {
                String[] data = line.split(",", -1);

                if (data.length == 7 &&
                        data[3].equals(String.valueOf(route))) {
                    System.out.println(line);
                }
            }
        }
    }

    public static void main(String[] args) throws IOException {

        addPass(new BusPass(
                "P1001", "192311001", "Arun Kumar",
                12, "Central Bus Stop",
                "2027-06-30", 4500.00));

        BusPass result = findPass("P1001");

        if (result != null)
            System.out.println("Found: " + result);
        else
            System.out.println("Pass not found");

        System.out.println("Passes on Route 12:");
        listByRoute(12);
    }
}

Observation:
•	Writing records in append mode is efficient.
•	Searching requires sequential scanning of records.
•	Listing records requires reading the complete file.
•	Execution time generally increases as the number of records increases.
Complexity Analysis:
